fetch("http://127.0.0.1:9000/insert", {
  method: "POST",
  headers: {
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    id: "1",
    value: "Ali"
  })
})
.then(res => res.text())
.then(data => console.log("INSERT:", data))
.catch(err => console.error(err));
